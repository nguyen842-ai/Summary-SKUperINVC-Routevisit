function doGet(e) { // Trả về định dạng JSON, hỗ trợ CORS var output = ContentService.createTextOutput(); output.setMimeType(ContentService.MimeType.JSON); try { // ID Sheet lấy từ URL của bạn var sheetId = '1Wz87qszHZmT-JhcU7SG2YLjOgKntRSroIR-U4Zuu5ys'; var ss = SpreadsheetApp.openById(sheetId); // Lấy tên nhân viên truyền vào từ tham số URL var kaeName = e.parameter.kaeName; if (!kaeName) { return output.setContent(JSON.stringify({ status: "error", message: "Thiếu tham số kaeName" })); } // Tìm sheet trùng với tên nhân viên đã xử lý var sheet = ss.getSheetByName(kaeName); if (!sheet) { return output.setContent(JSON.stringify({ status: "error", message: "Không tìm thấy sheet: " + kaeName })); } // Giả sử các dòng cảnh báo được viết ở Cột A. Lấy toàn bộ dữ liệu cột A. var data = sheet.getDataRange().getValues(); var comments = []; for (var i = 0; i < data.length; i++) { if (data[i][0] && String(data[i][0]).trim() !== "") { comments.push(String(data[i][0]).trim()); } } return output.setContent(JSON.stringify({ status: "success", comments: comments })); } catch (error) { return output.setContent(JSON.stringify({ status: "error", message: error.toString() })); } }
